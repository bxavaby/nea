    hl.exec_cmd("dbus-update-activation-environment --systemd DISPLAY WAYLAND_DISPLAY XDG_CURRENT_DESKTOP")

    -- Idle daemon 4 auto-locking & screen-off

    -- Wpp daemon

    -- Main
    hl.exec_cmd("waybar")
    -- hl.exec_cmd("kitty")
    -- hl.exec_cmd("rofi -show drun")
    hl.exec_cmd("nm-applet --indicator")

    -- Notifications
    hl.exec_cmd("mako")
end)

#!/usr/bin/env zsh
if [[ -z $1 ]]; then
  printf '%s\n' \
    "  Lock" \
    "⏾  Suspend" \
    "  Logout" \
    "  Reboot" \
    "  Shutdown"
  exit 0
fi
case "$1" in
